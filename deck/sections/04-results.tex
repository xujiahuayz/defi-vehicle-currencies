\begin{frame}{Fragmentation against succession}
\begin{columns}[T]
\begin{column}{0.34\textwidth}
\begin{itemize}
\item Effective vehicles \keyfig{1.42} to \keyfig{8.43} by value, \keyfig{1.86} to \keyfig{7.31} by count
\item Leader's share of value 83.7\% to 25.5\%, of count 71.5\% to 27.9\%
\item Leadership alternating from 2022 on value alone, native leads every year on count
\item Succession: concentration holds, the leader changes
\item Fragmentation: concentration falls, the incumbent may lead
\end{itemize}
