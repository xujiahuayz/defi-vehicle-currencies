\begin{frame}{A7. Daily and weekly clocks answer different questions}
\begin{enumerate}
\item \textbf{Primary panel}\hfill day observations; date fixed effects
\item \textbf{Calendar sensitivity}\hfill day observations; week and weekday fixed effects
\item \textbf{Aggregation robustness}\hfill complete weeks; week fixed effects
\item \textbf{Transition series}\hfill exact calendar links; HAC at 1, 7, 30, and 120 days
\end{enumerate}
\vspace{0.65cm}
{\small Weekly ratios aggregate raw numerators and denominators before division. All seven weekday anchors are reported.\par}
\end{frame}

\begin{frame}{A8. Count and value define separate dominance margins}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Count-weighted share}
\[
S^{N}_{k,t}=\frac{N^I_{k,t}}{\sum_j N^I_{j,t}}
\]
Each reconstructed route carries one unit of weight.
\end{column}
\begin{column}{0.46\textwidth}
\textbf{Value-weighted share}
\[
S^{V}_{k,t}=\frac{\mathrm{Vol}^I_{k,t}}{\sum_j \mathrm{Vol}^I_{j,t}}
\]
Each route carries supported realised USD value.
\end{column}
\end{columns}
\vspace{0.75cm}
{\small A larger value shift can reflect route size, routing topology, or candidate composition. The decomposition tests these margins directly.\par}
\end{frame}

\begin{frame}{A10. Capital, inventory, and depth are distinct quantities}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\textbf{Deposited capital}\par Funds committed to a pool; reconciled balances or audited provider TVL.
\vspace{0.45cm}
\textbf{Inventory}\par Exact token amounts held by the pool.
\end{column}
\begin{column}{0.47\textwidth}
\textbf{Executable depth}\par Output near a price under the invariant and exact quote state.
\vspace{0.45cm}
\textbf{Quote quality}\par Realised output for notional $q$ on executable paths.
\end{column}
\end{columns}
\vspace{0.65cm}
{\small TVL can measure capital. It is not automatically the denominator for executable depth.\par}
\end{frame}
