\begin{frame}{A9. Intermediary and route-endpoint use are separate}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Intermediary share}
\[
I^N_{k,t}=\frac{n^I_{k,t}}{\sum_j n^I_{j,t}}
\]
How often asset $k$ sits inside a route.
\end{column}
\begin{column}{0.46\textwidth}
\textbf{Endpoint share}
\[
E^N_{k,t}=\frac{n^E_{k,t}}{\sum_j n^E_{j,t}}
\]
How often $k$ appears at either observed route endpoint.
\end{column}
\end{columns}
\vspace{0.16cm}
{\small Observed pool-route endpoints supply the benchmark.\par}
\vspace{0.16cm}
{\normalsize\color{ucldark}\bfseries Excess use compares intermediation with observed route-endpoint use on the same support.\par}
\end{frame}
