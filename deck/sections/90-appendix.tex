\begin{frame}{A9. Route-endpoint and intermediary use are separate margins}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Intermediary share}
\[
I^N_{k,t}=\frac{n^I_{k,t}}{\sum_j n^I_{j,t}}
\]
How often asset $k$ sits inside a route.
\end{column}
\begin{column}{0.46\textwidth}
\textbf{Endpoint share}
\[
E^N_{k,t}=\frac{n^E_{k,t}}{\sum_j n^E_{j,t}}
\]
How often asset $k$ appears at either end of the observed pool route.
\end{column}
\end{columns}
\vspace{0.32cm}
{\small The benchmark describes reconstructed pool routes; an aggregator instruction can begin or end outside the connected component.\par}
\vspace{0.30cm}
{\large\color{ucldark}\bfseries Excess use compares intermediation with observed route-endpoint use on the same support.\par}
\end{frame}
