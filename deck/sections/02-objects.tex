\begin{frame}{Four measurements separate role, scale, and network position}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{enumerate}
\item \textbf{Vehicle share}\par Fraction of route units intermediated by asset $k$.
\vspace{0.35em}
\item \textbf{Excess use}\par Intermediary share relative to endpoint demand.
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\begin{enumerate}
\setcounter{enumi}{2}
\item \textbf{Network position}\par Degree, strength, and betweenness on the trading graph.
\vspace{0.35em}
\item \textbf{Execution advantage}\par Direct output relative to a vehicle route at one state.
\end{enumerate}
\end{column}
\end{columns}
\vspace{0.75cm}
{\large\color{ucldark}\bfseries A single statistic cannot stand in for all four objects.\par}
\end{frame}
