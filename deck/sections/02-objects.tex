% VISUAL-FUNCTION: measurement taxonomy | two-by-two conceptual grid | prevent vehicle use, excess use, topology, and cost from collapsing
\begin{frame}{Four measures capture use, scale, and network position}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{enumerate}
\item \textbf{Vehicle share}\par Fraction of route units intermediated by asset $k$.
\vspace{0.35em}
\item \textbf{Excess use}\par Intermediary share relative to endpoint demand.
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\begin{enumerate}
\setcounter{enumi}{2}
\item \textbf{Network position}\par Degree, strength, and betweenness on the trading graph.
\vspace{0.35em}
\item \textbf{Execution advantage}\par Direct-route output relative to a feasible vehicle route at the same pre-trade state.
\end{enumerate}
\end{column}
\end{columns}
\vspace{0.75cm}
{\large\color{ucldark}\bfseries Together, the measures distinguish use, scale, network position, and execution cost.\par}
\end{frame}
