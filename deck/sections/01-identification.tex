\begin{frame}{Measuring vehicle dominance}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.43\textwidth}
\evidencekicker{Measurement}
\begin{itemize}
\item Dominance is the share of indirect trade intermediated by asset $k$.
\item Count share treats each route equally; value share weights routes by dollar value.
\item Network measures capture how broadly $k$ connects endpoint pairs.
\end{itemize}

\begin{frame}{Protocol design first imposed ETH intermediation}
\begin{columns}[T]
\begin{column}{0.44\textwidth}
\evidencekicker{Institutional fact}
\begin{itemize}
\item Uniswap V1 created one exchange contract per token, paired with ETH.
\item Token-to-token swaps therefore moved through ETH by construction.
\item V2 allowed arbitrary pairs; the ETH-centred pairing network persisted.
\end{itemize}
\end{column}
