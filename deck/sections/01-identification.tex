\begin{frame}{Question: what makes a vehicle currency dominant?}
\vspace{0.2cm}
{\large One asset sits between many buyers and sellers. What gives it the largest share of intermediation?\par}
\vspace{0.65cm}
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\begin{enumerate}
\item The cheapest asset to hold and trade
\item The asset with the deepest pairing network
\item The asset embedded in routing and market design
\item Several assets, each dominant in a different margin
\end{enumerate}
\end{column}

\end{frame}

\begin{frame}{Vehicle status is binary; dominance is continuous}
\begin{columns}[T]
\begin{column}{0.43\textwidth}
\evidencekicker{Measurement}
\begin{itemize}
\item One route through asset $k$ gives it vehicle status.
\item Dominance measures the share of intermediation that $k$ captures.
\item Count and value answer different economic questions.
\end{itemize}

\begin{frame}{Protocol design first imposed ETH intermediation}
\begin{columns}[T]
\begin{column}{0.44\textwidth}
\evidencekicker{Institutional fact}
\begin{itemize}
\item Uniswap V1 created one exchange contract per token, paired with ETH.
\item Token-to-token swaps therefore moved through ETH by construction.
\item V2 allowed arbitrary pairs; the ETH-centred pairing network persisted.
\end{itemize}
\end{column}
